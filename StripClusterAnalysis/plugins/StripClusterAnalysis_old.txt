  g4SimHitsTIDLowTofToken_  = consumes<std::vector<PSimHit>>(iConfig.getParameter<edm::InputTag>("g4SimHitsTIDLowTofTag"));
  g4SimHitsTIDHighTofToken_ = consumes<std::vector<PSimHit>>(iConfig.getParameter<edm::InputTag>("g4SimHitsTIDHighTofTag"));

  clusterTree->Branch("hit_tid_lo_z",hit_tid_lo_z,"hit_tid_lo_z/F");

// ------------ method called for each event  ------------
void StripClusterAnalysis::analyze(const edm::Event& iEvent, const edm::EventSetup& iSetup) {
  using namespace edm;
  
  const auto& tkGeom = &iSetup.getData(tkGeomToken_);
  const auto tkDets = tkGeom->dets();

  edm::Handle<std::vector<PSimHit>> g4SimHitsTOBHighTofToken = iEvent.getHandle(g4SimHitsTOBHighTofToken_);
  edm::Handle<std::vector<PSimHit>> g4SimHitsTOBLowTofToken = iEvent.getHandle(g4SimHitsTOBLowTofToken_);
  edm::Handle<std::vector<PSimHit>> g4SimHitsTIBHighTofToken = iEvent.getHandle(g4SimHitsTIBHighTofToken_);
  edm::Handle<std::vector<PSimHit>> g4SimHitsTIBLowTofToken = iEvent.getHandle(g4SimHitsTIBLowTofToken_);
  edm::Handle<std::vector<PSimHit>> g4SimHitsTIDHighTofToken = iEvent.getHandle(g4SimHitsTIDHighTofToken_);
  edm::Handle<std::vector<PSimHit>> g4SimHitsTIDLowTofToken = iEvent.getHandle(g4SimHitsTIDLowTofToken_);
  edm::Handle<std::vector<PSimHit>> g4SimHitsTECHighTofToken = iEvent.getHandle(g4SimHitsTECHighTofToken_);
  edm::Handle<std::vector<PSimHit>> g4SimHitsTECLowTofToken = iEvent.getHandle(g4SimHitsTECLowTofToken_);

  edm::Handle<std::vector<SimTrack>> g4SimHitsTrack = iEvent.getHandle(g4SimHitsTrackToken_);

  edm::Handle<edmNew::DetSetVector<SiStripCluster>> clusterCollection = iEvent.getHandle(clusterToken_);

  eventN = iEvent.id().event();
  runN   = (int) event.id().run();
  lumi  = (int) event.id().luminosityBlock();

  if (!g4SimHitsTOBHighTofToken.isValid()){
    edm::LogError("TrackAnalyzer") << "No valid g4SimHits collection found";
    return;  
  }
  int nTrack = 0;
  for(const auto& simtrack : *g4SimHitsTrack){
    float px = simtrack.momentum().Px();
    float py = simtrack.momentum().Py();
    simtrk_pt = TMath::Sqrt(px*px + py*py);
    ++nTrack;
    
    int nTOBhits=0, nTIBhits=0, nTIDhits=0, nTEChits=0;
    for (const auto& simhit : *g4SimHitsTOBLowTofToken){
      auto tob_hit_gp = HitPosition(simhit,tkGeom);
      hit_tob_lo_x[nTOBhits] = tob_hit_gp.x();
      hit_tob_lo_y[nTOBhits] = tob_hit_gp.y();
      hit_tob_lo_z[nTOBhits] = tob_hit_gp.z();
      ++nTOBhits;
      //clusterTree->Fill();
    }
    for (const auto& simhit : *g4SimHitsTIBLowTofToken){
      auto tib_hit_gp = HitPosition(simhit,tkGeom);
      hit_tib_lo_x[nTIBhits] = tib_hit_gp.x();
      hit_tib_lo_y[nTIBhits] = tib_hit_gp.y();
      hit_tib_lo_z[nTIBhits] = tib_hit_gp.z();
      ++nTIBhits;
      //clusterTree->Fill();
    }
    for (const auto& simhit : *g4SimHitsTIDLowTofToken){
      auto tid_hit_gp = HitPosition(simhit,tkGeom);
      hit_tid_lo_x[nTIDhits] = tid_hit_gp.x();
      hit_tid_lo_y[nTIDhits] = tid_hit_gp.y();
      hit_tid_lo_z[nTIDhits] = tid_hit_gp.z();
      ++nTIDhits;
      //clusterTree->Fill();
    
    }
    for (const auto& simhit : *g4SimHitsTECLowTofToken){
      auto tec_hit_gp = HitPosition(simhit,tkGeom);
      hit_tec_lo_x[nTEChits] = tec_hit_gp.x();
      hit_tec_lo_y[nTEChits] = tec_hit_gp.y();
      hit_tec_lo_z[nTEChits] = tec_hit_gp.z();
      ++nTEChits;
      //clusterTree->Fill();
    
    }
    int tothits = nTOBhits+nTIBhits+nTIDhits+nTEChits;
    min_tothits = std::min(tothits,70);
    //std::cout<<min_tothits<<std::endl;
    if(nTrack > 0 && min_tothits > 0)
      nhitpTrack = min_tothits/nTrack;
    clusterTree->Fill();

  }


  for (const auto& detSiStripClusters : *clusterCollection) {
    uint32_t detId = detSiStripClusters.detId();
    uint16_t nStrips{0};
    
    for (const auto& stripCluster : detSiStripClusters) {
       uint16_t firstStrip  = stripCluster.firstStrip();
       uint16_t endStrip    = stripCluster.endStrip();
       const auto& _detId = detId; // for the capture clause in the lambda function
       auto det = std::find_if(tkDets.begin(), tkDets.end(), [_detId](auto& elem) -> bool {
	   return (elem->geographicalId().rawId() == _detId);
	 });
       const StripTopology& p = dynamic_cast<const StripGeomDetUnit*>(*det)->specificTopology();
       nStrips = p.nstrips() - 1;
       uint16_t barycenter = SiStripCluster(stripCluster,nStrips).barycenter();
       GlobalPoint gp = (tkGeom->idToDet(detId))->surface().toGlobal(p.localPosition((float) barycenter));
       striphit_x = gp.x();
       striphit_y = gp.y();
       striphit_z = gp.z();
								    
      // for (int strip = firstStrip; strip < endStrip+1; ++strip)
      // 	{
      // 	  GlobalPoint gp = (tkGeom->idToDet(detId))->surface().toGlobal(p.localPosition((float) strip));
      // 	  striphit_x[strip - firstStrip] = gp.x();
      // 	  striphit_y[strip - firstStrip] = gp.y();
      // 	  striphit_z[strip - firstStrip] = gp.z();
      // 	}
      clusterTree->Fill();
    }
  }
#ifdef THIS_IS_AN_EVENTSETUP_EXAMPLE
  // if the SetupData is always needed
  auto setup = iSetup.getData(setupToken_);
  // if need the ESHandle to check if the SetupData was there or not
  auto pSetup = iSetup.getHandle(setupToken_);
#endif
}
